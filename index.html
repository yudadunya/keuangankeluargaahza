// ====================== HELPER FUNCTIONS ====================== function parseAmount(text) { const lower = text.toLowerCase(); // Prioritas 1: Format dengan satuan (paling akurat) // Contoh: 15rb, 15 ribu, 1.5jt, 1,5 juta, 250k, 3.5jt const unitRegex = /(\d{1,3}(?:[.,]\d{3})*|\d+)([.,]\d+)?\s*(rb|ribu|k|jt|juta|j)\b/gi; let match; let candidates = []; while ((match = unitRegex.exec(lower)) !== null) { let numStr = match[1].replace(/\./g, '').replace(/,/g, ''); let decimal = match[2] ? match[2].replace(',', '.') : ''; let num = parseFloat(numStr + decimal); const unit = match[3].toLowerCase(); if (['rb', 'ribu', 'k'].includes(unit)) num *= 1000; if (['jt', 'juta', 'j'].includes(unit)) num *= 1000000; candidates.push(num); } if (candidates.length > 0) { return Math.max(...candidates); // Ambil yang terbesar (biasanya nominal uang) } // Prioritas 2: Angka biasa (tanpa satuan) // Abaikan angka kecil (1-30) yang kemungkinan kuantitas const numberRegex = /(\d{1,3}(?:[.,]\d{3})+|\d{4,})/g; // prioritaskan angka >= 1000 atau berformat ribuan const numbers = []; let m; while ((m = numberRegex.exec(text)) !== null) { let n = parseFloat(m[1].replace(/\./g, '').replace(/,/g, '')); if (n >= 1000) numbers.push(n); } if (numbers.length > 0) return Math.max(...numbers); // Prioritas 3: Angka kecil sebagai fallback (hati-hati) const smallNumbers = text.match(/\b(\d+)\b/g); if (smallNumbers) { const parsed = smallNumbers.map(n => parseFloat(n)).filter(n => n >= 100); // minimal 100 if (parsed.length > 0) return Math.max(...parsed); } return 0; } function detectAccounts(textLower) { const keywords = [ { key: 'bca syariah', name: 'BCA Syariah' }, { key: 'bca-syariah', name: 'BCA Syariah' }, { key: 'bsi', name: 'BSI' }, { key: 'bjb', name: 'BJB' }, { key: 'seabank', name: 'Seabank' }, { key: 'ajaib', name: 'Ajaib' }, { key: 'bca', name: 'BCA' }, { key: 'mandiri', name: 'Mandiri' }, { key: 'bri', name: 'BRI' }, { key: 'bni', name: 'BNI' }, { key: 'gopay', name: 'GoPay' }, { key: 'ovo', name: 'OVO' }, { key: 'shopeepay', name: 'ShopeePay' }, { key: 'dana', name: 'Dana' }, { key: 'dompet', name: 'Cash' }, { key: 'tunai', name: 'Cash' }, { key: 'cash', name: 'Cash' } ]; let found = []; keywords.forEach(item => { if (textLower.includes(item.key) && !found.includes(item.name)) { found.push(item.name); } }); return found; } function detectDate(textLower) { const now = new Date(); let date = new Date(now); if (textLower.includes('kemarin')) { date.setDate(date.getDate() - 1); } else if (textLower.includes('lusa')) { date.setDate(date.getDate() + 2); } else if (textLower.includes('besok')) { date.setDate(date.getDate() + 1); } else { // Coba deteksi "tanggal 5" atau "tgl 12" const tglMatch = textLower.match(/(?:tanggal|tgl|tgl\.)\s*(\d{1,2})/); if (tglMatch) { const day = parseInt(tglMatch[1]); if (day >= 1 && day <= 31) { date.setDate(day); } } } return `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, '0')}-${String(date.getDate()).padStart(2, '0')}`; } function detectCategory(lower, type) { if (lower.match(/makan|beli|ayam|bawang|sayur|beras|dapur|minum|nutrijel|gula|susu|tahu|tempe|lauk|nasi|goreng|bakso|mie|kopi|teh/)) { return 'Konsumsi & Dapur'; } if (lower.match(/bensin|transport|parkir|gojek|grab|ojek|tol|angkot/)) { return 'Transportasi'; } if (lower.match(/spp|sekolah|buku|kursus|les|uang saku anak/)) { return 'Pendidikan Anak'; } if (lower.match(/listrik|air|wifi|pulsa|token|internet|tagihan/)) { return 'Utilitas & Tagihan'; } if (lower.match(/infaq|sedekah|amal|zakat|ngasih|sumbangan/)) { return 'Sosial & Infaq'; } if (lower.match(/obat|dokter|rumah sakit|apotek|vitamin/)) { return 'Kesehatan'; } if (lower.match(/baju|celana|sepatu|belanja|shop|online/)) { return 'Belanja'; } if (type === 'Pemasukan') { if (lower.match(/gaji|omset|jual|terima|pendapatan|bonus/)) return 'Inflow / Pendapatan'; return 'Pemasukan Lain'; } return 'Umum'; } function cleanDescription(text, amount, accounts) { let desc = text .replace(/pemasukan|pengeluaran|masuk|keluar|rp\.?|transfer dari|transfer ke|transfer|pindah|dari|ke/gi, '') .replace(/bsi|bjb|seabank|bca syariah|bca|ajaib|mandiri|bri|bni|gopay|ovo|dana|shopeepay|cash|tunai|dompet/gi, '') .replace(/\b\d+([.,]\d+)?\s*(rb|ribu|k|jt|juta|j)\b/gi, '') .replace(/\b\d{1,3}([.,]\d{3})+\b/g, '') .replace(/\b\d+\b/g, '') .replace(/\s+/g, ' ') .trim(); if (!desc || desc.length < 2) { desc = text.substring(0, 60); } return desc; } // ====================== MAIN PROCESS ====================== async function processAICommand(text) { const lower = text.toLowerCase().trim(); const amount = parseAmount(text); const accounts = detectAccounts(lower); const primaryAccount = accounts.length > 0 ? accounts[0] : 'Cash'; const tanggal = detectDate(lower); const formatIDR = (val) => new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR' }).format(val); // 1. Koreksi Saldo const isBalanceAdjustment = (lower.includes('saldo') || lower.includes('set saldo') || lower.includes('update saldo')) && accounts.length > 0 && amount > 0; if (isBalanceAdjustment) { const { data: allTx } = await supabaseClient.from('transaksi').select('*'); let currentAccBalance = 0; if (allTx) { allTx.forEach(row => { let jml = parseFloat(row.jumlah) || 0; let acc = row.akun || 'Cash'; let isMasuk = (row.tipe || '').toLowerCase().includes('masuk'); if (acc === primaryAccount) { if (isMasuk) currentAccBalance += jml; else currentAccBalance -= jml; } }); } let selisih = amount - currentAccBalance; if (Math.abs(selisih) < 1) { return `ℹ️ Saldo ${primaryAccount} sudah sesuai: ${formatIDR(amount)}`; } let adjType = selisih > 0 ? 'Pemasukan' : 'Pengeluaran'; let adjAmount = Math.abs(selisih); const { error } = await supabaseClient.from('transaksi').insert([{ tipe: adjType, kategori: 'Koreksi / Penyesuaian Saldo', akun: primaryAccount, keterangan: `Koreksi saldo ${primaryAccount} menjadi ${formatIDR(amount)}`, jumlah: adjAmount, tanggal: tanggal }]); if (error) return `❌ Gagal menyesuaikan saldo: ${error.message}`; return `✅ Saldo ${primaryAccount} diperbarui
• Saldo baru: ${formatIDR(amount)}
• Penyesuaian: ${adjType} ${formatIDR(adjAmount)}`; } // 2. Minta Rekap const isReport = (lower.includes('rekap') || lower.includes('laporan') || lower === 'saldo' || lower.includes('total')) && amount === 0; if (isReport) { return await fetchAndShowSummary(); } // 3. Validasi nominal if (amount <= 0) { return `⚠️ Saya belum menemukan nominal yang jelas.
Coba ketik lebih spesifik, contoh:
• "beli bawang 15rb"
• "gaji masuk 5jt ke bca"
• "transfer 200rb dari bsi ke bjb"`; } // 4. Deteksi Transfer const isTransfer = (lower.includes('transfer') || lower.includes('pindah') || lower.includes('kirim') || lower.includes('dari') && lower.includes('ke')) && accounts.length >= 2; if (isTransfer) { // Coba deteksi urutan "dari A ke B" let sourceAcc = accounts[0]; let destAcc = accounts[1]; const dariIndex = lower.indexOf('dari'); const keIndex = lower.indexOf('ke'); if (dariIndex > -1 && keIndex > dariIndex) { // Urutan sudah benar di keyword } else if (keIndex > -1 && dariIndex > keIndex) { // Terbalik sourceAcc = accounts[1]; destAcc = accounts[0]; } const { error: err1 } = await supabaseClient.from('transaksi').insert([{ tipe: 'Pengeluaran', kategori: 'Transfer Internal', akun: sourceAcc, keterangan: `Transfer ke ${destAcc}`, jumlah: amount, tanggal: tanggal }]); const { error: err2 } = await supabaseClient.from('transaksi').insert([{ tipe: 'Pemasukan', kategori: 'Transfer Internal', akun: destAcc, keterangan: `Terima dari ${sourceAcc}`, jumlah: amount, tanggal: tanggal }]); if (err1 || err2) return `❌ Gagal memproses transfer.`; return `🔄 Transfer Berhasil (${tanggal})
• Dari: ${sourceAcc} (-${formatIDR(amount)})
• Ke: ${destAcc} (+${formatIDR(amount)})`; } // 5. Transaksi biasa let type = 'Pengeluaran'; if (lower.match(/masuk|pemasukan|terima|jual|omset|gaji|bonus|pendapatan|dapat/)) { type = 'Pemasukan'; } const category = detectCategory(lower, type); const description = cleanDescription(text, amount, accounts); const { error } = await supabaseClient.from('transaksi').insert([{ tipe: type, kategori: category, akun: primaryAccount, keterangan: description, jumlah: amount, tanggal: tanggal }]); if (error) return `❌ Gagal menyimpan: ${error.message}`; const badgeColor = type === 'Pemasukan' ? 'text-emerald-400' : 'text-rose-400'; return `✅ Tercatat (${tanggal})
• Tipe: ${type}
• Akun: ${primaryAccount}
• Kategori: ${category}
• Jumlah: ${formatIDR(amount)}
• Keterangan: ${description}`; }
